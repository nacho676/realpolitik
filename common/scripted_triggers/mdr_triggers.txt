# The AI buys ammunition abroad when a war is draining its stockpile.
mdr_ammo_ai_should_buy = {
	has_war = yes
	check_variable = { mdr_ammo_days_left < 28 }
	check_variable = { mdr_ammo_k < mdr_ammo_capacity_k }
}

# Country scope: random disasters can hit this country now (rule and cooldown). The historical pause is per state (mdr_hz_historical).
mdr_disaster_can_strike = {
	NOT = { has_game_rule = { rule = mdr_rule_disasters option = disabled } }
	NOT = { has_country_flag = mdr_disaster_cooldown }
}

# State scope. Hurricane and typhoon season: June to November, or December to April in the southern hemisphere.
mdr_disaster_state_storm_season = {
	OR = {
		AND = {
			NOT = { has_state_flag = mdr_hz_south }
			check_variable = { global.month > 5 }
			check_variable = { global.month < 12 }
		}
		AND = {
			has_state_flag = mdr_hz_south
			OR = {
				check_variable = { global.month = 12 }
				check_variable = { global.month < 5 }
			}
		}
	}
}

# State scope. Summer, when wildfires and heatwaves peak: June to September, or December to February in the south.
mdr_disaster_state_summer = {
	OR = {
		AND = {
			NOT = { has_state_flag = mdr_hz_south }
			check_variable = { global.month > 5 }
			check_variable = { global.month < 10 }
		}
		AND = {
			has_state_flag = mdr_hz_south
			OR = {
				check_variable = { global.month = 12 }
				check_variable = { global.month < 3 }
			}
		}
	}
}

# State scope. Tornado season: March to July, or October to January in the south.
mdr_disaster_state_spring = {
	OR = {
		AND = {
			NOT = { has_state_flag = mdr_hz_south }
			check_variable = { global.month > 2 }
			check_variable = { global.month < 8 }
		}
		AND = {
			has_state_flag = mdr_hz_south
			OR = {
				check_variable = { global.month > 9 }
				check_variable = { global.month < 2 }
			}
		}
	}
}

# THIS the possible buyer, ROOT the seller: at war with us, in a faction at war with us, or thinks very badly of us.
mdr_sale_is_rival = {
	OR = {
		has_war_with = ROOT
		any_enemy_country = { is_in_faction_with = ROOT }
		has_opinion = { target = ROOT value < -50 }
	}
}

# Country scope: owns at least one naval base.
mdr_sale_has_port = {
	any_owned_state = {
		any_province_building_level = {
			province = { all_provinces = yes }
			building = naval_base
			level > 0
		}
	}
}

# Mongolia's pan-Mongol path (mdr_mongolia.txt): a neighbour in civil war, defeat, collapse or exhaustion.
mdr_mon_country_weak = {
	OR = {
		has_civil_war = yes
		has_capitulated = yes
		has_stability < 0.2
		AND = { has_war = yes surrender_progress > 0.3 }
		check_variable = { mdr_wex > 20 }
	}
}
