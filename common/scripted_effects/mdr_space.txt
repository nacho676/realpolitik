# mdr_space_budget: hundredths of a percent of GDP a year (0 to 500). Technologies: common/technologies/mdr_space.txt (generated).
# Tab in Millennium Dawn's Space Programme: common/scripted_guis/mdr_space_gui.txt, interface/mdr_space.gui.

# Country init: who had what in 2000.

	else_if = { limit = { original_tag = UKR } set_variable = { mdr_space_budget = 14 } }

	else_if = { limit = { original_tag = BRA } set_variable = { mdr_space_budget = 2 } }

	else_if = { limit = { original_tag = BEL } set_variable = { mdr_space_budget = 6 } }
	else_if = { limit = { original_tag = SWE } set_variable = { mdr_space_budget = 3 } }
	else_if = { limit = { original_tag = SWI } set_variable = { mdr_space_budget = 3 } }
	else_if = { limit = { original_tag = NRY } set_variable = { mdr_space_budget = 3 } }
	else_if = { limit = { original_tag = DEN } set_variable = { mdr_space_budget = 1 } }

	if = {

			OR = {
				original_tag = FRA original_tag = GER original_tag = ITA original_tag = ENG original_tag = SPR original_tag = HOL
				original_tag = BEL original_tag = SWE original_tag = SWI original_tag = NRY original_tag = DEN original_tag = AUS

	mdr_space_refresh = yes

# Country scope: figures for the menu and the modifier.
mdr_space_refresh = {
	mdr_space_count_techs = yes
	set_variable = { mdr_space_sites = 0 }
	set_variable = { mdr_space_site_levels = 0 }
	clear_array = mdr_space_site_states
	every_controlled_state = {
		if = { limit = { check_variable = { building_level@mdr_spaceport > 0 } }
			add_to_array = { PREV.mdr_space_site_states = THIS.id }
			add_to_variable = { PREV.mdr_space_sites = 1 }
			add_to_variable = { PREV.mdr_space_site_levels = building_level@mdr_spaceport }
		}
	}
	# The agency grows: its own launcher and a base make it level 2; heavy launchers and its own crewed capsule, level 3.
	if = { limit = { check_variable = { mdr_space_tier = 1 } has_tech = mdr_sp_launcher check_variable = { mdr_space_sites > 0 } }
		set_variable = { mdr_space_tier = 2 }
	}
	if = { limit = { check_variable = { mdr_space_tier = 2 } has_tech = mdr_sp_heavy has_tech = mdr_sp_crewed has_country_flag = mdr_sa_crew_capsule_done }
		set_variable = { mdr_space_tier = 3 }
	}
	# What the levels give, for the menu. The modifiers themselves are in the building.
	set_variable = { mdr_space_site_speed_pct = mdr_space_site_levels }
	multiply_variable = { mdr_space_site_speed_pct = 4.3 }
	set_variable = { mdr_space_site_reliability_pct = mdr_space_site_levels }
	multiply_variable = { mdr_space_site_reliability_pct = 1.7 }
	set_variable = { mdr_space_site_cost_pct = mdr_space_site_levels }
	multiply_variable = { mdr_space_site_cost_pct = 1.2 }
	# Russia leases Baikonur while Kazakhstan holds it: it is not one of its own, and only its science counts.
	set_variable = { mdr_space_leased_levels = 0 }
	if = { limit = { original_tag = SOV }
		717 = {
			if = { limit = { is_controlled_by = KAZ check_variable = { building_level@mdr_spaceport > 0 } }
				add_to_variable = { PREV.mdr_space_leased_levels = building_level@mdr_spaceport }
			}
		}
	}
	set_variable = { mdr_space_year_cost = gdp_total }
	multiply_variable = { mdr_space_year_cost = mdr_space_budget }
	divide_variable = { mdr_space_year_cost = 10000 }
	set_temp_variable = { mdr_space_cost_factor = 1 }
	add_to_temp_variable = { mdr_space_cost_factor = modifier@mdr_space_budget_cost_factor }
	multiply_variable = { mdr_space_year_cost = mdr_space_cost_factor }
	mdr_space_standing_update = yes
	set_variable = { mdr_space_week_cost = mdr_space_year_cost }
	divide_variable = { mdr_space_week_cost = 52 }
	set_variable = { mdr_space_budget_week = mdr_space_week_cost }
	add_to_variable = { mdr_space_week_cost = mdr_space_payroll }
	set_variable = { mdr_space_chance_pct = modifier@olv_launch_reliability_modifier }
	add_to_variable = { mdr_space_chance_pct = 0.84 }
	if = { limit = { check_variable = { mdr_space_chance_pct > 0.97 } } set_variable = { mdr_space_chance_pct = 0.97 } }
	multiply_variable = { mdr_space_chance_pct = 100 }
	set_variable = { mdr_space_hq_science_cost = mdr_space_hq }
	multiply_variable = { mdr_space_hq_science_cost = 40 }
	set_variable = { mdr_space_hq_money_cost = mdr_space_hq }
	multiply_variable = { mdr_space_hq_money_cost = 1.2 }
	set_variable = { mdr_st_olv_c = mdr_st_debris_c }
	add_to_variable = { mdr_st_olv_c = mdr_sm_olv_c }
	add_to_variable = { mdr_st_olv_c = mdr_pc_olv_c }
	set_variable = { mdr_st_budget = mdr_sm_budget }
	add_to_variable = { mdr_st_budget = mdr_pc_budget }
	# The bar has a frame for each tenth of a percent.
	set_variable = { mdr_space_budget_frame = mdr_space_budget }
	divide_variable = { mdr_space_budget_frame = 10 }
	round_variable = mdr_space_budget_frame
	add_to_variable = { mdr_space_budget_frame = 1 }
	set_variable = { mdr_space_budget_pct = mdr_space_budget }
	divide_variable = { mdr_space_budget_pct = 100 }
	# What the budget is worth, in tenths of a percent: above 1% of GDP each tenth returns 0.35 of a normal one.
	set_variable = { mdr_space_budget_worth = mdr_space_budget }
	divide_variable = { mdr_space_budget_worth = 10 }
	if = { limit = { check_variable = { mdr_space_budget > 100 } }
		set_temp_variable = { mdr_space_budget_extra = mdr_space_budget_worth }
		subtract_from_temp_variable = { mdr_space_budget_extra = 10 }
		multiply_temp_variable = { mdr_space_budget_extra = 0.35 }
		set_variable = { mdr_space_budget_worth = 10 }
		add_to_variable = { mdr_space_budget_worth = mdr_space_budget_extra }
	}
	# Science: 0.7 points a week per tenth of a percent and agency level, plus 0.8 per level of launch base, and what the technologies add.
	set_variable = { mdr_space_science_week = mdr_space_budget_worth }
	multiply_variable = { mdr_space_science_week = mdr_space_tier }
	multiply_variable = { mdr_space_science_week = 0.7 }
	set_temp_variable = { mdr_space_site_gain = 0.8 }
	add_to_temp_variable = { mdr_space_site_gain = modifier@mdr_space_site_science }
	set_temp_variable = { mdr_space_site_count = mdr_space_site_levels }
	add_to_temp_variable = { mdr_space_site_count = mdr_space_leased_levels }
	multiply_temp_variable = { mdr_space_site_gain = mdr_space_site_count }
	set_variable = { mdr_space_site_science_week = mdr_space_site_gain }
	add_to_variable = { mdr_space_science_week = mdr_space_site_gain }
	# Every agency has a staff that produces something, whatever its size.
	add_to_variable = { mdr_space_science_week = 1.4 }
	set_temp_variable = { mdr_space_science_mult = 1 }
	add_to_temp_variable = { mdr_space_science_mult = modifier@mdr_space_science_factor }
	set_temp_variable = { mdr_space_hq_bonus = mdr_space_hq }
	multiply_temp_variable = { mdr_space_hq_bonus = 0.05 }
	add_to_temp_variable = { mdr_space_science_mult = mdr_space_hq_bonus }
	multiply_variable = { mdr_space_science_week = mdr_space_science_mult }
	# Spin-offs: +2% research and +10% launch base construction at 1% of GDP, +4.8% and +24% at 5%.
	set_variable = { mdr_space_bonus_research = mdr_space_budget_worth }
	multiply_variable = { mdr_space_bonus_research = 0.002 }
	set_variable = { mdr_space_bonus_sites = mdr_space_budget_worth }
	multiply_variable = { mdr_space_bonus_sites = 0.01 }
	set_variable = { mdr_space_bonus_research_pct = mdr_space_budget_worth }
	multiply_variable = { mdr_space_bonus_research_pct = 0.2 }
	set_variable = { mdr_space_bonus_sites_pct = mdr_space_budget_worth }
	if = { limit = { check_variable = { mdr_space_tier > 0 } check_variable = { mdr_space_budget > 0 } }
		if = { limit = { NOT = { has_dynamic_modifier = { modifier = mdr_space_modifier } } }
			add_dynamic_modifier = { modifier = mdr_space_modifier }
		}
	}
	else_if = { limit = { has_dynamic_modifier = { modifier = mdr_space_modifier } }
		remove_dynamic_modifier = { modifier = mdr_space_modifier }
	}
	mdr_space_spend_chart = yes
	if = { limit = { is_ai = no } add_to_variable = { global.international_systems_updated = 1 } }
}

# Country scope, weekly: adds science and counts the launch bases. The budget is paid as an additional expense (00_money_system.txt).
mdr_space_weekly = {
	if = { limit = { NOT = { has_variable = mdr_space_tier } } mdr_space_start = yes }
	if = { limit = { check_variable = { mdr_space_tier > 0 } }
		mdr_space_program_weekly = yes
		mdr_space_spend_track = yes
		mdr_space_refresh = yes
		if = { limit = { check_variable = { mdr_space_budget > 0 } }
			add_to_variable = { mdr_space_science = mdr_space_science_week }
		}
	}
	else_if = { limit = { is_ai = no } mdr_space_refresh = yes }
	# The AI founds its own agency when it is rich enough, after some three years on average.
	else_if = {
		limit = {
			has_tech = mdr_sp_foundations
			check_variable = { gdp_total > 200 }
			check_variable = { treasury > 10 }
			has_political_power > 96
		}
		# About one week in 170: some three years.
		random_list = {
			6 = { mdr_space_create = yes }
			994 = { }
		}
	}
}

# World, weekly, with the historical rule: agencies founded after 2000, if the country can pay for one.
mdr_space_historical = {
	if = { limit = { has_game_rule = { rule = mdr_rule_world_events option = historical } }
		if = { limit = { date > 2004.02.01 NOT = { has_global_flag = mdr_space_hist_per } }
			set_global_flag = mdr_space_hist_per
			if = { limit = { country_exists = PER } PER = { mdr_space_historical_found = yes } }
		}
		if = { limit = { date > 2014.07.22 NOT = { has_global_flag = mdr_space_hist_uae } }
			set_global_flag = mdr_space_hist_uae
			if = { limit = { country_exists = UAE } UAE = { mdr_space_historical_found = yes } }
		}
		if = { limit = { date > 2018.07.01 NOT = { has_global_flag = mdr_space_hist_ast } }
			set_global_flag = mdr_space_hist_ast
			if = { limit = { country_exists = AST } AST = { mdr_space_historical_found = yes } }
		}
		if = { limit = { date > 2018.12.13 NOT = { has_global_flag = mdr_space_hist_tur } }
			set_global_flag = mdr_space_hist_tur
			if = { limit = { country_exists = TUR } TUR = { mdr_space_historical_found = yes } }
		}
	}
}
