# Natural disasters and accidents.
# Types (mdr_disaster_type): 1 earthquake, 2 tsunami, 3 tropical storm, 4 flood, 5 wildfire, 6 heatwave, 7 industrial accident,
# 8 nuclear accident (historical only), 9 tornado.
# Tiers (mdr_disaster_tier): 1 mild, 2 intermediate, 3 severe (catastrophic).
#
# Random disasters follow geography, not borders. Every state takes the hazard profile of its strategic region
# (map/strategicregions) once per game. Each month every state rolls against its profile for the season; the type comes
# from the same weights, then the tier. Whoever controls the state suffers it, so a conquered region keeps its hazards.

# Once per game: the hazard profile of every state, from its strategic region.
mdr_disaster_zones_start = {
	if = { limit = { NOT = { has_global_flag = mdr_disaster_zones_added } }
		set_global_flag = mdr_disaster_zones_added
		every_state = { mdr_disaster_state_profile = yes }
	}
}

				region = 160 region = 161 region = 165 region = 174 region = 181 region = 188 region = 190 region = 195 region = 203
			}
		}
		set_state_flag = mdr_hz_historical

	set_variable = { mdr_hz_q = mdr_p_q }

# State scope. Outputs: temp variables mdr_w_* (this month's weight of each hazard here) and mdr_w_total.
mdr_disaster_state_weights = {
	set_temp_variable = { mdr_w_quake = mdr_hz_q }
	set_temp_variable = { mdr_w_tsunami = 0 }
	set_temp_variable = { mdr_w_storm = 0 }
	if = { limit = { is_coastal = yes }
		set_temp_variable = { mdr_w_tsunami = mdr_hz_t }
		if = { limit = { mdr_disaster_state_storm_season = yes } set_temp_variable = { mdr_w_storm = mdr_hz_s } }
	}
	set_temp_variable = { mdr_w_flood = mdr_hz_f }
	if = { limit = { check_variable = { global.month > 5 } check_variable = { global.month < 10 } }
		add_to_temp_variable = { mdr_w_flood = mdr_hz_m }
	}
	# Wildfires, heatwaves and tornadoes are four or five times rarer out of season.
	set_temp_variable = { mdr_w_fire = mdr_hz_w }
	set_temp_variable = { mdr_w_heat = mdr_hz_h }
	if = { limit = { NOT = { mdr_disaster_state_summer = yes } }
		divide_temp_variable = { mdr_w_fire = 4 }
		divide_temp_variable = { mdr_w_heat = 4 }
	}
	set_temp_variable = { mdr_w_tornado = mdr_hz_o }
	if = { limit = { NOT = { mdr_disaster_state_spring = yes } } divide_temp_variable = { mdr_w_tornado = 5 } }
	# Global warming (mdr_climate.txt): more storms, floods, wildfires and heatwaves, at most +30%.
	if = { limit = { check_variable = { global.mdr_climate_factor > 1 } }
		multiply_temp_variable = { mdr_w_storm = global.mdr_climate_factor }
		multiply_temp_variable = { mdr_w_flood = global.mdr_climate_factor }
		multiply_temp_variable = { mdr_w_fire = global.mdr_climate_factor }
		multiply_temp_variable = { mdr_w_heat = global.mdr_climate_factor }
	}

	set_temp_variable = { mdr_w_total = mdr_w_quake }
	add_to_temp_variable = { mdr_w_total = mdr_w_tsunami }
	add_to_temp_variable = { mdr_w_total = mdr_w_storm }
	add_to_temp_variable = { mdr_w_total = mdr_w_flood }
	add_to_temp_variable = { mdr_w_total = mdr_w_fire }
	add_to_temp_variable = { mdr_w_total = mdr_w_heat }
	add_to_temp_variable = { mdr_w_total = mdr_w_tornado }
}

# Monthly, once for the whole world. A state rolls its weights against 50,000: a Japanese coastal state in typhoon season
# (about 130) has a 0.26% chance a month, a German state (about 30) 0.06%.
mdr_disasters_world_monthly = {
	every_state = {
		limit = {
			has_variable = mdr_hz_f
			# Antarctica has no controller.
			OR = {
				NOT = { has_state_flag = mdr_polar }
				any_country = { controls_state = PREV }
			}
			CONTROLLER = { mdr_disaster_can_strike = yes }
			# Historical mode: the regions with real disasters wait until 2027, whoever owns them.
			OR = {
				NOT = { has_state_flag = mdr_hz_historical }
				has_game_rule = { rule = mdr_rule_disasters option = dynamic }
				date > 2026.12.31
			}
		}
		mdr_disaster_state_weights = yes
		random_list = {
			50000 = { }
			mdr_w_total = {
				set_temp_variable = { mdr_dis_type = 0 }
				CONTROLLER = { save_event_target_as = mdr_disaster_controller }
				save_event_target_as = mdr_disaster_state
				mdr_disaster_pick_type = yes
				mdr_disaster_roll_tier = yes
				event_target:mdr_disaster_controller = { mdr_disaster_strike = yes }
			}
		}
	}
}

# Mild disasters are common, intermediate ones less so, severe ones rare. Output: temp variable mdr_dis_tier.

# State scope. Input: temp variables mdr_dis_type and mdr_dis_tier.
# Whoever controls the state takes the spirit and the cost, so occupied land hits the occupier.
# With temp variable mdr_dis_mute = 1 nothing happens: the start-of-game pass only marks past dates as done.
mdr_disaster_hit_state = {
	if = { limit = { NOT = { check_variable = { mdr_dis_mute = 1 } } }
		save_event_target_as = mdr_disaster_state
		controller = {
			mdr_disaster_strike = yes
		}
	}
}

# State scope. Extra damage for multi-region historical disasters, without a second event.
mdr_disaster_damage_state = {
	if = { limit = { NOT = { check_variable = { mdr_dis_mute = 1 } } }
		controller = {
			if = { limit = { check_variable = { gdp_per_capita < 7 } } set_temp_variable = { mdr_dis_wealth = 3 } }
			else_if = { limit = { check_variable = { gdp_per_capita < 20 } } set_temp_variable = { mdr_dis_wealth = 2 } }
			else = { set_temp_variable = { mdr_dis_wealth = 1 } }
		}
		mdr_disaster_damage = yes
	}
}

# Country scope (the victim). Inputs: temp variables mdr_dis_type and mdr_dis_tier, event target mdr_disaster_state.
mdr_disaster_strike = {
	set_variable = { mdr_disaster_type = mdr_dis_type }
	set_variable = { mdr_disaster_tier = mdr_dis_tier }

	# Building codes and emergency services matter: poor countries suffer more.
	if = { limit = { check_variable = { gdp_per_capita < 7 } } set_temp_variable = { mdr_dis_wealth = 3 } }
	else_if = { limit = { check_variable = { gdp_per_capita < 20 } } set_temp_variable = { mdr_dis_wealth = 2 } }
	else = { set_temp_variable = { mdr_dis_wealth = 1 } }

	# Reconstruction cost: 0.2%, 1% or 3% of GDP, scaled down for rich economies where one region is a small share
	# (Katrina cost the US about 1% of GDP; the 2010 earthquake cost Haiti more than its whole GDP).
	set_variable = { mdr_disaster_cost = gdp_total }
	if = { limit = { check_variable = { mdr_dis_tier = 1 } } multiply_variable = { mdr_disaster_cost = 0.002 } }
	else_if = { limit = { check_variable = { mdr_dis_tier = 2 } } multiply_variable = { mdr_disaster_cost = 0.01 } }
	else = { multiply_variable = { mdr_disaster_cost = 0.03 } }
	if = { limit = { check_variable = { mdr_dis_wealth = 1 } } multiply_variable = { mdr_disaster_cost = 0.4 } }
	else_if = { limit = { check_variable = { mdr_dis_wealth = 2 } } multiply_variable = { mdr_disaster_cost = 0.7 } }

	# Climate adaptation technologies: less cost from the climate's disasters, and a chance of lighter damage (mdr_disaster_damage).
	set_temp_variable = { mdr_dis_resil = 0 }
	if = { limit = { check_variable = { mdr_dis_type > 2 } check_variable = { mdr_dis_type < 7 } }
		set_temp_variable = { mdr_dis_resil = modifier@mdr_disaster_resilience }
		set_temp_variable = { mdr_dis_cost_mult = 1 }
		subtract_from_temp_variable = { mdr_dis_cost_mult = mdr_dis_resil }
		multiply_variable = { mdr_disaster_cost = mdr_dis_cost_mult }
	}

	remove_ideas = {
		mdr_disaster_moderate
		mdr_disaster_severe
		mdr_disaster_catastrophe
	}
	if = { limit = { check_variable = { mdr_dis_tier = 1 } }
		add_timed_idea = { idea = mdr_disaster_moderate days = 30 }
		set_country_flag = { flag = mdr_disaster_cooldown days = 90 value = 1 }
	}
	else_if = { limit = { check_variable = { mdr_dis_tier = 2 } }
		add_timed_idea = { idea = mdr_disaster_severe days = 90 }
		set_country_flag = { flag = mdr_disaster_cooldown days = 365 value = 1 }
	}
	else_if = { limit = { check_variable = { mdr_dis_type = 8 } }
		add_timed_idea = { idea = mdr_disaster_nuclear days = 365 }
		set_country_flag = { flag = mdr_disaster_cooldown days = 365 value = 1 }
	}
	else = {
		add_timed_idea = { idea = mdr_disaster_catastrophe days = 180 }
		set_country_flag = { flag = mdr_disaster_cooldown days = 365 value = 1 }
	}

	event_target:mdr_disaster_state = { mdr_disaster_damage = yes }

	# Lost lives and displaced workers.
	if = { limit = { check_variable = { mdr_dis_tier = 1 } }
		if = { limit = { check_variable = { mdr_dis_wealth = 1 } } add_manpower = -200 }
		else_if = { limit = { check_variable = { mdr_dis_wealth = 2 } } add_manpower = -500 }
		else = { add_manpower = -1000 }
	}
	else_if = { limit = { check_variable = { mdr_dis_tier = 2 } }
		if = { limit = { check_variable = { mdr_dis_wealth = 1 } } add_manpower = -2000 }
		else_if = { limit = { check_variable = { mdr_dis_wealth = 2 } } add_manpower = -5000 }
		else = { add_manpower = -10000 }
	}
	else = {
		if = { limit = { check_variable = { mdr_dis_wealth = 1 } } add_manpower = -10000 }
		else_if = { limit = { check_variable = { mdr_dis_wealth = 2 } } add_manpower = -25000 }
		else = { add_manpower = -60000 }
	}

	if = { limit = { check_variable = { mdr_dis_type = 1 } } country_event = { id = mdr_disasters.1 } }
	else_if = { limit = { check_variable = { mdr_dis_type = 2 } } country_event = { id = mdr_disasters.2 } }
	else_if = { limit = { check_variable = { mdr_dis_type = 3 } } country_event = { id = mdr_disasters.3 } }
	else_if = { limit = { check_variable = { mdr_dis_type = 4 } } country_event = { id = mdr_disasters.4 } }
	else_if = { limit = { check_variable = { mdr_dis_type = 5 } } country_event = { id = mdr_disasters.5 } }
	else_if = { limit = { check_variable = { mdr_dis_type = 6 } } country_event = { id = mdr_disasters.6 } }
	else_if = { limit = { check_variable = { mdr_dis_type = 7 } } country_event = { id = mdr_disasters.7 } }
	else_if = { limit = { check_variable = { mdr_dis_type = 9 } } country_event = { id = mdr_disasters.9 } }
	else = { country_event = { id = mdr_disasters.8 } }

	if = { limit = { check_variable = { mdr_dis_tier = 3 } }
		set_global_flag = { flag = mdr_catastrophe_recent days = 30 value = 1 }
		save_event_target_as = mdr_disaster_country_news
		event_target:mdr_disaster_state = { save_event_target_as = mdr_disaster_state_news }
		if = { limit = { check_variable = { mdr_dis_type = 1 } } news_event = { id = mdr_disasters.11 } }
		else_if = { limit = { check_variable = { mdr_dis_type = 2 } } news_event = { id = mdr_disasters.12 } }
		else_if = { limit = { check_variable = { mdr_dis_type = 3 } } news_event = { id = mdr_disasters.13 } }
		else_if = { limit = { check_variable = { mdr_dis_type = 4 } } news_event = { id = mdr_disasters.14 } }
		else_if = { limit = { check_variable = { mdr_dis_type = 5 } } news_event = { id = mdr_disasters.15 } }
		else_if = { limit = { check_variable = { mdr_dis_type = 6 } } news_event = { id = mdr_disasters.16 } }
		else_if = { limit = { check_variable = { mdr_dis_type = 7 } } news_event = { id = mdr_disasters.17 } }
		else_if = { limit = { check_variable = { mdr_dis_type = 9 } } news_event = { id = mdr_disasters.19 } }
		else = { news_event = { id = mdr_disasters.18 } }
		# The main powers decide whether to send aid. PREV is the victim inside the loop.
		save_event_target_as = mdr_disaster_victim
		every_country = {
			limit = {
				mdr_is_aid_donor = yes
				NOT = { tag = PREV }
				NOT = { has_war_with = PREV }
			}
			country_event = { id = mdr_disasters.20 }
		}
	}
	log = "[GetDateText]: [This.GetName]: MDR disaster type [?mdr_dis_type] tier [?mdr_dis_tier] in [mdr_disaster_state.GetName]"
}

# State scope. Inputs: temp variables mdr_dis_tier and mdr_dis_wealth.
mdr_disaster_damage = {
	# An early warning center, and dikes against water, each lower the damage one step.
	set_temp_variable = { mdr_dis_local = mdr_dis_wealth }
	if = { limit = { check_variable = { building_level@mdr_early_warning_center > 0 } }
		subtract_from_temp_variable = { mdr_dis_local = 1 }
	}
	if = {
		limit = {
			any_province_building_level = {
				province = { all_provinces = yes }
				building = mdr_coastal_dikes
				level > 0
			}
			OR = {
				check_variable = { mdr_dis_type = 2 }
				check_variable = { mdr_dis_type = 3 }
				check_variable = { mdr_dis_type = 4 }
			}
		}
		subtract_from_temp_variable = { mdr_dis_local = 1 }
	}
	if = { limit = { check_variable = { mdr_dis_resil > 0 } }
		set_temp_variable = { mdr_dis_roll = random }
		if = { limit = { check_variable = { mdr_dis_roll < mdr_dis_resil } } subtract_from_temp_variable = { mdr_dis_local = 1 } }
	}
	if = { limit = { check_variable = { mdr_dis_local < 1 } }
		damage_building = { type = infrastructure damage = 0.3 }
	}
	else = {
		if = { limit = { check_variable = { mdr_dis_tier = 1 } }
			if = { limit = { check_variable = { mdr_dis_local = 1 } } damage_building = { type = infrastructure damage = 0.3 } }
			else_if = { limit = { check_variable = { mdr_dis_local = 2 } } damage_building = { type = infrastructure damage = 0.5 } }
			else = { damage_building = { type = infrastructure damage = 0.8 } }
		}
		else_if = { limit = { check_variable = { mdr_dis_tier = 2 } }
			if = { limit = { check_variable = { mdr_dis_local = 1 } }
				damage_building = { type = infrastructure damage = 1 }
				damage_building = { type = industrial_complex damage = 0.5 }
			}
			else_if = { limit = { check_variable = { mdr_dis_local = 2 } }
				damage_building = { type = infrastructure damage = 1.5 }
				damage_building = { type = industrial_complex damage = 1 }
			}
			else = {
				damage_building = { type = infrastructure damage = 2 }
				damage_building = { type = industrial_complex damage = 1.5 }
			}
		}
		else = {
			if = { limit = { check_variable = { mdr_dis_local = 1 } }
				damage_building = { type = infrastructure damage = 2 repair_speed_modifier = -0.25 }
				damage_building = { type = industrial_complex damage = 1.5 repair_speed_modifier = -0.25 }
				damage_building = { type = arms_factory damage = 1 repair_speed_modifier = -0.25 }
			}
			else_if = { limit = { check_variable = { mdr_dis_local = 2 } }
				damage_building = { type = infrastructure damage = 3 repair_speed_modifier = -0.25 }
				damage_building = { type = industrial_complex damage = 2.5 repair_speed_modifier = -0.25 }
				damage_building = { type = arms_factory damage = 1.5 repair_speed_modifier = -0.25 }
			}
			else = {
				damage_building = { type = infrastructure damage = 4 repair_speed_modifier = -0.25 }
				damage_building = { type = industrial_complex damage = 3.5 repair_speed_modifier = -0.25 }
				damage_building = { type = arms_factory damage = 2 repair_speed_modifier = -0.25 }
			}
		}
	}
}

# Each relief option steps the spirit down one level.
mdr_disaster_downgrade = {
	if = {
		limit = {
			OR = {
				has_idea = mdr_disaster_catastrophe
				has_idea = mdr_disaster_nuclear
			}
		}
		remove_ideas = {
			mdr_disaster_catastrophe
			mdr_disaster_nuclear
		}
		add_timed_idea = { idea = mdr_disaster_severe days = 90 }
	}
	else_if = { limit = { has_idea = mdr_disaster_severe }
		remove_ideas = mdr_disaster_severe
		add_timed_idea = { idea = mdr_disaster_moderate days = 30 }
	}
	else_if = { limit = { has_idea = mdr_disaster_moderate }
		remove_ideas = mdr_disaster_moderate
	}
}

# Weekly, once for the whole world. Real disasters of 2000 to 2025 on their dates.
mdr_disasters_historical = {
	if = { limit = { date > 2001.01.12 NOT = { has_global_flag = mdr_hd_2001_el_salvador } }
