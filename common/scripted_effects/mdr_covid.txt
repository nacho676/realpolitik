# Historical COVID-19. COVID_events.0 fires on the owner of Hubei (state 570) in the first week of December 2019.
# Each country gets MD's COVID_events.4 (lockdown and COVID_pandemic_impact) on the date its outbreak hit hard.
# Day 0 is about 5 December 2019. COVID_events.4 already skips the origin country.
mdr_covid_launch = {
	set_global_flag = mdr_covid_launched
	random_country = {
		limit = { owns_state = 570 }
		country_event = { id = COVID_events.0 }
	}
	log = "[GetDateText]: MDR historical COVID launched"
}

	# Everyone else between mid-March and late April 2020.
	every_country = {
		limit = {
			has_capitulated = no
			NOT = {
				OR = {
					tag = KOR
					tag = PER
					tag = ITA
					tag = JAP
					tag = TUR
					tag = USA
					tag = SPR
					tag = CAN
					tag = FRA
					tag = AST
					tag = CHL
					tag = ARG
					tag = BRA
					tag = GER
					tag = ENG
					tag = COL
					tag = RAJ
					tag = EGY
					tag = SAF
					tag = MEX
					tag = SOV
				}
			}
		}
		country_event = { id = COVID_events.4 days = 100 random_days = 40 }
	}

	# Late March 2020, when factory closures and port shutdowns peaked.
	news_event = { id = COVID_events.5 days = 115 }
}

# Mass vaccination from mid-December 2020. Rich countries clear the pandemic first,
# middle-income ones in spring 2021, the rest wait for the spirit to expire.
mdr_covid_vaccines = {
	set_global_flag = mdr_covid_vaccines_started
	every_country = {
		limit = {
			OR = {
				has_idea = COVID_pandemic_impact
				has_idea = COVID_pandemic_origin
			}
		}
		if = {
			limit = { check_variable = { gdp_per_capita > 19.999 } }
			country_event = { id = mdr_covid.1 days = 1 random_days = 60 }
		}
		else_if = {
			limit = { check_variable = { gdp_per_capita > 6.999 } }
			country_event = { id = mdr_covid.1 days = 90 random_days = 90 }
		}
	}
	log = "[GetDateText]: MDR COVID vaccination campaigns scheduled"
}
