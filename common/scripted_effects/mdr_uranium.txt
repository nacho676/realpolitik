# Uranium is a real ninth resource (common/resources/mdr_resources.txt). The engine trades it and the AI buys it.
# Each Millennium Dawn enrichment facility consumes 2 uranium and nuclear missile lines 1 per factory; without it, reactor fuel output falls.

# The mines working in 2000, in the states of the largest producers. Runs once at game start.
# Levels follow WNA 2000 output (tU): Canada 10,683 = level 10, then about one level per 1,000 t, at least 1.
# Each mine level yields 5 uranium, +1 per mining technology (common/technologies/mdr_uranium_techs.txt); mines can be built in any state, each level costing more than the last.
mdr_uranium_start = {
	if = { limit = { NOT = { has_global_flag = mdr_uranium_deposits_added } }
		set_global_flag = mdr_uranium_deposits_added
		756 = { add_building_construction = { type = mdr_uranium_mine level = 10 instant_build = yes } }
		745 = { add_building_construction = { type = mdr_uranium_mine level = 4 instant_build = yes } }
		747 = { add_building_construction = { type = mdr_uranium_mine level = 4 instant_build = yes } }
		717 = { add_building_construction = { type = mdr_uranium_mine level = 2 instant_build = yes } }
		1108 = { add_building_construction = { type = mdr_uranium_mine level = 4 instant_build = yes } }
		282 = { add_building_construction = { type = mdr_uranium_mine level = 3 instant_build = yes } }
		1070 = { add_building_construction = { type = mdr_uranium_mine level = 4 instant_build = yes } }
		726 = { add_building_construction = { type = mdr_uranium_mine level = 3 instant_build = yes } }
		789 = { add_building_construction = { type = mdr_uranium_mine level = 2 instant_build = yes } }
		275 = { add_building_construction = { type = mdr_uranium_mine level = 1 instant_build = yes } }
		1090 = { add_building_construction = { type = mdr_uranium_mine level = 1 instant_build = yes } }
		562 = { add_building_construction = { type = mdr_uranium_mine level = 1 instant_build = yes } }
		451 = { add_building_construction = { type = mdr_uranium_mine level = 1 instant_build = yes } }

		# Small producers in 2000: Czech Republic (Rozna), Romania (Crucea), Pakistan (Dera Ghazi Khan), Texas (in-situ leaching).
		117 = { add_building_construction = { type = mdr_uranium_mine level = 1 instant_build = yes } }

	}
}

mdr_uranium_weekly = {
	# Demand from enrichment facilities: 2 uranium each.
	set_variable = { mdr_uranium_need = total_enrichment_facilities }
	multiply_variable = { mdr_uranium_need = 2 }
	# Only shown for countries with enrichment facilities.
	if = { limit = { check_variable = { mdr_uranium_need > 0 } NOT = { has_dynamic_modifier = { modifier = mdr_uranium_modifier } } }
		add_dynamic_modifier = { modifier = mdr_uranium_modifier }
	}
	else_if = { limit = { check_variable = { mdr_uranium_need < 1 } has_dynamic_modifier = { modifier = mdr_uranium_modifier } }
		remove_dynamic_modifier = { modifier = mdr_uranium_modifier }
	}
	# What is left after the facilities take their share; negative means a deficit.
	set_variable = { mdr_uranium_shortfall = 0 }
	if = {
		limit = {
			check_variable = { mdr_uranium_need > 0 }
			check_variable = { resource@uranium < 0 }
		}
		set_variable = { mdr_uranium_shortfall = resource@uranium }
		divide_variable = { mdr_uranium_shortfall = mdr_uranium_need }
		clamp_variable = { var = mdr_uranium_shortfall min = -1 max = 0 }
	}
	set_variable = { mdr_uranium_produced = resource_produced@uranium }
	set_variable = { mdr_uranium_imported = resource_imported@uranium }
	set_variable = { mdr_uranium_exported = resource_exported@uranium }

	# Mines and the states working them, for the uranium panel of the trade screen.
	set_temp_variable = { mdr_uranium_mines_t = 0 }
	set_temp_variable = { mdr_uranium_deposits_t = 0 }
	every_controlled_state = {
		limit = { check_variable = { building_level@mdr_uranium_mine > 0 } }
		add_to_temp_variable = { mdr_uranium_deposits_t = 1 }
		add_to_temp_variable = { mdr_uranium_mines_t = building_level@mdr_uranium_mine }
	}
	set_variable = { mdr_uranium_mines = mdr_uranium_mines_t }
	set_variable = { mdr_uranium_deposits = mdr_uranium_deposits_t }

	# Export revenue with Millennium Dawn's formula for the other resources (calculate_resource_sale_rate):
	# exported x price x export multiplier x 0.00298 per week. Multiplied first, divided last, for the 3-decimal precision.
	set_variable = { mdr_uranium_income = 0 }
	if = { limit = { check_variable = { mdr_uranium_exported > 0 } }
		set_variable = { mdr_uranium_income = mdr_uranium_exported }
		multiply_variable = { mdr_uranium_income = global.mdr_uranium_price }
		set_temp_variable = { mdr_uranium_export_mult = modifier@resource_export_multiplier_modifier }
		add_to_temp_variable = { mdr_uranium_export_mult = 1 }
		multiply_variable = { mdr_uranium_income = mdr_uranium_export_mult }
		multiply_variable = { mdr_uranium_income = 298 }
		divide_variable = { mdr_uranium_income = 100000 }
		set_temp_variable = { treasury_change = mdr_uranium_income }
		modify_treasury_effect = yes
	}
}

# Once a week for the whole world: price = 2.35 x demand / supply, between 0.7 and 11.75.
# 2.35 is the 2000 price (about 8 dollars a pound) on Millennium Dawn's scale (dollars a tonne / 7,500, as steel or aluminium);
# the 2007 spike reached about five times that.
mdr_uranium_world_market = {
	set_temp_variable = { mdr_uranium_supply = 0 }
	set_temp_variable = { mdr_uranium_demand = 0 }
	every_country = {
		add_to_temp_variable = { mdr_uranium_supply = mdr_uranium_produced }
		add_to_temp_variable = { mdr_uranium_demand = resource_consumed@uranium }
	}
	set_variable = { global.mdr_uranium_supply = mdr_uranium_supply }
	set_variable = { global.mdr_uranium_demand = mdr_uranium_demand }
	set_variable = { global.mdr_uranium_price = 2.35 }
	if = { limit = { check_variable = { mdr_uranium_supply > 0 } }
		set_temp_variable = { mdr_uranium_ratio = mdr_uranium_demand }
		divide_temp_variable = { mdr_uranium_ratio = mdr_uranium_supply }
		multiply_variable = { global.mdr_uranium_price = mdr_uranium_ratio }
	}
	clamp_variable = { var = global.mdr_uranium_price min = 0.7 max = 11.75 }
	# Market shocks (mdr_shocks.txt): the 2007 boom, Fukushima.
	if = { limit = { check_variable = { global.mdr_shock_uranium > 0 } } multiply_variable = { global.mdr_uranium_price = global.mdr_shock_uranium } }
}

# Kazakhstan becomes the largest producer in the world after 2009.
mdr_uranium_historical = {
	if = { limit = { date > 2009.01.01 NOT = { has_global_flag = mdr_hu_2009_kazakhstan } }
		set_global_flag = mdr_hu_2009_kazakhstan
		717 = {
			add_building_construction = {
				type = mdr_uranium_mine
				level = 2
				instant_build = yes
			}
		}
	}
}

# Country scope, once at game start: the countries that already mined with modern methods in 2000.
mdr_uranium_techs_start = {
	if = { limit = { OR = { original_tag = SOV original_tag = CHI original_tag = USA original_tag = CAN original_tag = AST original_tag = FRA } }
		set_technology = { mdr_uranium_tech_1 = 1 popup = no }
		mdr_uranium_tech_bonus = yes
	}
}

# Country scope. Called when a uranium mining technology is researched: a fixed +1, which the state's infrastructure does not multiply.
mdr_uranium_tech_bonus = {
	modify_building_resources = {
		building = mdr_uranium_mine
		resource = uranium
		amount = 1
	}
}
