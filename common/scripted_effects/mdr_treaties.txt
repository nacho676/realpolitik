# Arms control treaties. Membership is a country flag mdr_treaty_<id>:
# 1 NPT, 2 CTBT, 3 START, 4 INF, 5 ABM, 6 CFE, 7 Open Skies, 8 Ottawa (landmines), 9 CCM (cluster munitions), 10 JCPOA,
# 11 Kyoto Protocol, 12 Paris Agreement (climate: mdr_climate.txt).
# START, INF and ABM bind only the United States and Russia; JCPOA binds Iran.

mdr_treaties_start = {
	if = { limit = { NOT = { OR = { original_tag = RAJ original_tag = PAK original_tag = ISR original_tag = CUB } } }
		set_country_flag = mdr_treaty_1
	}
	if = { limit = { NOT = { OR = { original_tag = RAJ original_tag = PAK original_tag = NKO original_tag = SAU original_tag = SYR } } }
		set_country_flag = mdr_treaty_2
	}
	if = { limit = { OR = { original_tag = USA original_tag = SOV } }
		set_country_flag = mdr_treaty_3
		set_country_flag = mdr_treaty_4
		set_country_flag = mdr_treaty_5
	}
	if = { limit = { mdr_is_cfe_party = yes }
		set_country_flag = mdr_treaty_6
		set_country_flag = mdr_treaty_7
	}
	if = { limit = { NOT = { OR = { original_tag = USA original_tag = SOV original_tag = CHI original_tag = RAJ original_tag = PAK original_tag = ISR original_tag = KOR original_tag = NKO original_tag = PER original_tag = EGY original_tag = SYR original_tag = CUB original_tag = VIE original_tag = LBA original_tag = FIN original_tag = POL original_tag = SAU original_tag = UAE original_tag = MOR original_tag = LIT original_tag = LAT original_tag = EST } } }
		set_country_flag = mdr_treaty_8
	}
	add_dynamic_modifier = { modifier = mdr_treaties_modifier }
	mdr_treaties_recalc = yes
}

mdr_treaties_recalc = {
	set_variable = { mdr_tr_reactor = 0 }
	set_variable = { mdr_tr_nukes = 0 }
	set_variable = { mdr_tr_pp = 0 }
	set_variable = { mdr_tr_defence = 0 }
	set_variable = { mdr_tr_attack = 0 }
	set_variable = { mdr_tr_productivity = 0 }
	set_variable = { mdr_tr_investment = 0 }
	# NPT members get civil nuclear cooperation.
	if = { limit = { has_country_flag = mdr_treaty_1 } add_to_variable = { mdr_tr_reactor = 0.10 } }
	# START caps warheads.
	if = { limit = { has_country_flag = mdr_treaty_3 } add_to_variable = { mdr_tr_nukes = -0.25 } }
	# Each superpower arms-control treaty keeps detente alive.
	if = { limit = { has_country_flag = mdr_treaty_3 } add_to_variable = { mdr_tr_pp = 0.02 } }
	if = { limit = { has_country_flag = mdr_treaty_4 } add_to_variable = { mdr_tr_pp = 0.02 } }
	if = { limit = { has_country_flag = mdr_treaty_5 } add_to_variable = { mdr_tr_pp = 0.02 } }
	# Outside Ottawa an army can mine its borders; outside the CCM it can use cluster munitions.
	if = { limit = { NOT = { has_country_flag = mdr_treaty_8 } } add_to_variable = { mdr_tr_defence = 0.01 } }
	if = { limit = { NOT = { has_country_flag = mdr_treaty_9 } } add_to_variable = { mdr_tr_attack = 0.01 } }
	# JCPOA: sanctions relief in exchange for limits on enrichment.
	if = { limit = { has_country_flag = mdr_treaty_10 }
		add_to_variable = { mdr_tr_reactor = -0.30 }
		add_to_variable = { mdr_tr_productivity = 0.01 }
		add_to_variable = { mdr_tr_investment = -0.10 }
	}
	if = { limit = { NOT = { has_dynamic_modifier = { modifier = mdr_treaties_modifier } } }
		add_dynamic_modifier = { modifier = mdr_treaties_modifier }
	}
}

# Input: temp variable mdr_treaty_id.
mdr_treaty_join = {
	if = { limit = { check_variable = { mdr_treaty_id = 1 } } set_country_flag = mdr_treaty_1 }
	else_if = { limit = { check_variable = { mdr_treaty_id = 2 } } set_country_flag = mdr_treaty_2 }
	else_if = { limit = { check_variable = { mdr_treaty_id = 3 } } set_country_flag = mdr_treaty_3 }
	else_if = { limit = { check_variable = { mdr_treaty_id = 4 } } set_country_flag = mdr_treaty_4 }
	else_if = { limit = { check_variable = { mdr_treaty_id = 5 } } set_country_flag = mdr_treaty_5 }
	else_if = { limit = { check_variable = { mdr_treaty_id = 6 } } set_country_flag = mdr_treaty_6 }
	else_if = { limit = { check_variable = { mdr_treaty_id = 7 } } set_country_flag = mdr_treaty_7 }
	else_if = { limit = { check_variable = { mdr_treaty_id = 8 } } set_country_flag = mdr_treaty_8 }
	else_if = { limit = { check_variable = { mdr_treaty_id = 9 } } set_country_flag = mdr_treaty_9 }
	# Climate treaties: members pay for their targets for a while (mdr_climate_weekly).
	else_if = { limit = { check_variable = { mdr_treaty_id = 11 } }
		set_country_flag = mdr_treaty_11
		set_country_flag = { flag = mdr_kyoto_cost days = 2555 value = 1 }
	}
	else_if = { limit = { check_variable = { mdr_treaty_id = 12 } }
		set_country_flag = mdr_treaty_12
		set_country_flag = { flag = mdr_paris_cost days = 3650 value = 1 }
	}
	else = {
		set_country_flag = mdr_treaty_10
		set_global_flag = mdr_jcpoa_active
	}
	mdr_treaties_recalc = yes
}

# Input: temp variable mdr_treaty_id. Members resent the withdrawal and world tension rises.
mdr_treaty_leave = {
	if = { limit = { check_variable = { mdr_treaty_id = 1 } }
		clr_country_flag = mdr_treaty_1
		every_other_country = { limit = { has_country_flag = mdr_treaty_1 } add_opinion_modifier = { target = PREV modifier = mdr_left_arms_treaty } }
		add_threat = 5
	}
	else_if = { limit = { check_variable = { mdr_treaty_id = 2 } }
		clr_country_flag = mdr_treaty_2
		every_other_country = { limit = { has_country_flag = mdr_treaty_2 } add_opinion_modifier = { target = PREV modifier = mdr_left_arms_treaty } }
		add_threat = 3
	}
	else_if = { limit = { check_variable = { mdr_treaty_id = 6 } }
		clr_country_flag = mdr_treaty_6
		every_other_country = { limit = { has_country_flag = mdr_treaty_6 } add_opinion_modifier = { target = PREV modifier = mdr_left_arms_treaty } }
		add_threat = 2
	}
	else_if = { limit = { check_variable = { mdr_treaty_id = 7 } }
		clr_country_flag = mdr_treaty_7
		every_other_country = { limit = { has_country_flag = mdr_treaty_7 } add_opinion_modifier = { target = PREV modifier = mdr_left_arms_treaty } }
		add_threat = 1
	}
	else_if = { limit = { check_variable = { mdr_treaty_id = 8 } }
		clr_country_flag = mdr_treaty_8
		every_other_country = { limit = { has_country_flag = mdr_treaty_8 } add_opinion_modifier = { target = PREV modifier = mdr_left_arms_treaty } }
		add_threat = 1
	}
	else_if = { limit = { check_variable = { mdr_treaty_id = 9 } }
		clr_country_flag = mdr_treaty_9
		every_other_country = { limit = { has_country_flag = mdr_treaty_9 } add_opinion_modifier = { target = PREV modifier = mdr_left_arms_treaty } }
		add_threat = 1
	}
	# Leaving a climate treaty angers its members but does not raise world tension.
	else_if = { limit = { check_variable = { mdr_treaty_id = 11 } }
		clr_country_flag = mdr_treaty_11
		clr_country_flag = mdr_kyoto_cost
		every_other_country = { limit = { has_country_flag = mdr_treaty_11 } add_opinion_modifier = { target = PREV modifier = mdr_left_climate_treaty } }
	}
	else_if = { limit = { check_variable = { mdr_treaty_id = 12 } }
		clr_country_flag = mdr_treaty_12
		clr_country_flag = mdr_paris_cost
		every_other_country = { limit = { has_country_flag = mdr_treaty_12 } add_opinion_modifier = { target = PREV modifier = mdr_left_climate_treaty } }
	}
	# Bilateral treaties die for both parties.
	else_if = { limit = { check_variable = { mdr_treaty_id = 3 } }
		every_country = {
			limit = { has_country_flag = mdr_treaty_3 }
			clr_country_flag = mdr_treaty_3
			mdr_treaties_recalc = yes
			if = { limit = { NOT = { tag = PREV } } add_opinion_modifier = { target = PREV modifier = mdr_left_arms_treaty } }
		}
		add_threat = 5
	}
	else_if = { limit = { check_variable = { mdr_treaty_id = 4 } }
		every_country = {
			limit = { has_country_flag = mdr_treaty_4 }
			clr_country_flag = mdr_treaty_4
			mdr_treaties_recalc = yes
			if = { limit = { NOT = { tag = PREV } } add_opinion_modifier = { target = PREV modifier = mdr_left_arms_treaty } }
		}
		add_threat = 4
	}
	else_if = { limit = { check_variable = { mdr_treaty_id = 5 } }
		every_country = {
			limit = { has_country_flag = mdr_treaty_5 }
			clr_country_flag = mdr_treaty_5
			mdr_treaties_recalc = yes
			if = { limit = { NOT = { tag = PREV } } add_opinion_modifier = { target = PREV modifier = mdr_left_arms_treaty } }
		}
		add_threat = 3
	}
	else = {
		clr_global_flag = mdr_jcpoa_active
		every_country = {
			limit = { has_country_flag = mdr_treaty_10 }
			clr_country_flag = mdr_treaty_10
			mdr_treaties_recalc = yes
			if = { limit = { NOT = { tag = PREV } } add_opinion_modifier = { target = PREV modifier = mdr_left_arms_treaty } }
		}
		add_threat = 3
	}
	set_country_flag = { flag = mdr_treaty_cooldown days = 365 value = 1 }
	mdr_treaties_recalc = yes
	save_global_event_target_as = mdr_treaty_news_country
	set_variable = { global.mdr_treaty_news_id = mdr_treaty_id }
	news_event = { id = mdr_treaties.4 }
}

# Countries in the CFE and Open Skies regimes in 2000.
mdr_treaties_monthly = {
	# Now and then the international community presses holdouts to sign.
	if = {
		limit = {
			NOT = { has_country_flag = mdr_treaty_invite_cooldown }
			OR = {
				NOT = { has_country_flag = mdr_treaty_1 }
				NOT = { has_country_flag = mdr_treaty_2 }
				NOT = { has_country_flag = mdr_treaty_8 }
			}
		}
		random_list = {
			97 = { }
			3 = {
				set_country_flag = { flag = mdr_treaty_invite_cooldown days = 730 value = 1 }
				if = { limit = { NOT = { has_country_flag = mdr_treaty_1 } } set_variable = { mdr_treaty_offer = 1 } }
				else_if = { limit = { NOT = { has_country_flag = mdr_treaty_2 } } set_variable = { mdr_treaty_offer = 2 } }
				else = { set_variable = { mdr_treaty_offer = 8 } }
				country_event = { id = mdr_treaties.1 }
			}
		}
	}
}

# Historical treaty milestones. Actor countries run by a human get the choice in an event.
# Input: temp variable mdr_hist_id, actor country scope.
mdr_treaty_hist_action = {
	if = { limit = { NOT = { check_variable = { mdr_dis_mute = 1 } } }
		set_variable = { mdr_hist_action = mdr_hist_id }
		if = { limit = { is_ai = yes }
			mdr_treaty_hist_apply = yes
		}
		else = {
			country_event = { id = mdr_treaties.3 }
		}
	}
}

# Reads country variable mdr_hist_action.
