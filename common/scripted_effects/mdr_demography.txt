# Countries that aged fast after 2000, in the global weekly pass.
mdr_demography_historical = {
	if = { limit = { date > 2010.01.01 NOT = { has_global_flag = mdr_hdemo_2010_korea } }
		set_global_flag = mdr_hdemo_2010_korea
		if = { limit = { country_exists = KOR KOR = { has_idea = mdr_aging_1 } }
			KOR = { swap_ideas = { remove_idea = mdr_aging_1 add_idea = mdr_aging_2 } }
		}
	}
	if = { limit = { date > 2015.01.01 NOT = { has_global_flag = mdr_hdemo_2015_east_asia } }
		set_global_flag = mdr_hdemo_2015_east_asia
		if = { limit = { country_exists = CHI CHI = { has_idea = mdr_aging_1 } }
			CHI = { swap_ideas = { remove_idea = mdr_aging_1 add_idea = mdr_aging_2 } }
		}
		if = { limit = { country_exists = TAI TAI = { has_idea = mdr_aging_1 } }
			TAI = { swap_ideas = { remove_idea = mdr_aging_1 add_idea = mdr_aging_2 } }
		}
	}
	if = { limit = { date > 2020.01.01 NOT = { has_global_flag = mdr_hdemo_2020_korea } }
		set_global_flag = mdr_hdemo_2020_korea
		if = { limit = { country_exists = KOR KOR = { has_idea = mdr_aging_2 } }
			KOR = { swap_ideas = { remove_idea = mdr_aging_2 add_idea = mdr_aging_3 } }
		}
	}
}

# A finished reform path (common/national_focus/mdr_demography.txt) lowers aging one level.
mdr_demography_path_complete = {
	if = { limit = { has_idea = mdr_aging_3 }
		swap_ideas = { remove_idea = mdr_aging_3 add_idea = mdr_aging_2 }
	}
	else_if = { limit = { has_idea = mdr_aging_2 }
		swap_ideas = { remove_idea = mdr_aging_2 add_idea = mdr_aging_1 }
	}
	else_if = { limit = { has_idea = mdr_aging_1 }
		remove_ideas = mdr_aging_1
	}
}

# Input: temp variable mdr_demo_cost_permille (thousandths of GDP, paid once).
# Variables keep only three decimals, so multiply first and divide last.
mdr_demography_pay = {
	set_temp_variable = { treasury_change = gdp_total }
	multiply_temp_variable = { treasury_change = mdr_demo_cost_permille }
	divide_temp_variable = { treasury_change = -1000 }
	modify_treasury_effect = yes
}
