# When the AI builds the Realpolitik buildings (common/buildings/mdr_buildings.txt).

# Desalination where poor coastal soil keeps the country short of food.
mdr_ai_build_desalination = {
	enable = {
		check_variable = { mdr_food_self_sufficiency < 100 }
		any_owned_state = {
			is_controlled_by = ROOT
			is_coastal = yes
			OR = {
				has_state_flag = mdr_soil_poor
				NOT = { has_state_flag = mdr_arable_land }
			}
			NOT = { check_variable = { building_level@mdr_desalination_plant > 0 } }
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = building_target
		id = mdr_desalination_plant
		value = 4
	}
}

# Disaster protection where the risk is real (state hazard profile from mdr_disaster_state_profile, 0 to 45), only with
# factories to spare. building_target is a total for the country, so these stay small.
mdr_ai_build_early_warning = {
	enable = {
		NOT = { has_game_rule = { rule = mdr_rule_disasters option = disabled } }
		num_of_civilian_factories > 9
		num_of_available_civilian_factories > 3
		any_owned_state = {
			is_controlled_by = ROOT
			OR = {
				check_variable = { mdr_hz_q > 29 }
				check_variable = { mdr_hz_s > 34 }
				check_variable = { mdr_hz_t > 24 }
			}
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = building_target
		id = mdr_early_warning_center
		value = 2
	}
}
