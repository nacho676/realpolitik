# Top bar ammunition, like the game's fuel: "Full", days until full, or days until empty in red; the amount when balanced.
defined_text = {
	name = mdr_ammo_topbar_text
	text = {
		trigger = { NOT = { check_variable = { mdr_ammo_k < mdr_ammo_capacity_k } } NOT = { check_variable = { mdr_ammo_net_k < 0 } } }
		localization_key = MDR_AMMO_TOPBAR_FULL
	}
	text = {
		trigger = { check_variable = { mdr_ammo_net_k < 0 } }
		localization_key = MDR_AMMO_TOPBAR_EMPTYING
	}
	text = {
		trigger = { check_variable = { mdr_ammo_net_k > 0 } }
		localization_key = MDR_AMMO_TOPBAR_FILLING
	}
	text = {
		trigger = { check_variable = { mdr_ammo_k > 999.9 } }
		localization_key = MDR_AMMO_TOPBAR_M
	}
	text = {
		localization_key = MDR_AMMO_TOPBAR_K
	}
}
