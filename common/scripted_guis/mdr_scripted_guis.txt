scripted_gui = {
	# Prices of uranium, grain and rare earths. Hidden on the Economy tab, where Millennium Dawn's economy view covers the trade screen.
	mdr_tradeview_market_prices = {
		window_name = "mdr_tradeview_market_prices"
		context_type = player_context
		parent_window_token = trade_tab

		visible = {
			NOT = { check_variable = { MD_tradeview_tab_economy_state = 1 } }
		}

		ai_enabled = { always = no }
	}

	mdr_uranium_button = {
		window_name = "mdr_uranium_button_window"
		context_type = player_context
		parent_window_token = trade_tab

		effects = {
			mdr_uranium_open_button_click = {
				if = { limit = { has_country_flag = mdr_uranium_panel_open }
					clr_country_flag = mdr_uranium_panel_open
				}
				else = {
					set_country_flag = mdr_uranium_panel_open
					# Both panels open in the same place: close Millennium Dawn's resource storage panel.
					if = { limit = { has_country_flag = open_storage_screen }
						clr_country_flag = open_storage_screen
						update_dirty_resource_storage_var = yes
					}
				}
			}
		}

		ai_enabled = { always = no }
	}

		window_name = "mdr_uranium_panel"
