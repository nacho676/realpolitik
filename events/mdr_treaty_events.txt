	}

		text = mdr_treaties.1.d8
		trigger = { check_variable = { mdr_treaty_offer = 8 } }
	}
	picture = GFX_politics_negotiations
	is_triggered_only = yes

	option = {
		name = mdr_treaties.1.a
		log = "[GetDateText]: [This.GetName]: mdr_treaties.1.a executed"
		set_temp_variable = { mdr_treaty_id = mdr_treaty_offer }
		mdr_treaty_join = yes
		ai_chance = {
			base = 2
			modifier = {
				factor = 0
				check_variable = { mdr_treaty_offer = 1 }
				OR = {
					original_tag = RAJ
					original_tag = PAK
					original_tag = ISR
					original_tag = NKO
				}
			}
		}
	}

	option = {
		name = mdr_treaties.1.b
		log = "[GetDateText]: [This.GetName]: mdr_treaties.1.b executed"
		ai_chance = { base = 3 }
	}
}

# A superpower proposes an arms control treaty (FROM is the proposer)
country_event = {
	id = mdr_treaties.2
	title = mdr_treaties.2.t
	desc = {
		text = mdr_treaties.2.d3
		trigger = { check_variable = { mdr_treaty_proposal = 3 } }
	}
	desc = {
		text = mdr_treaties.2.d4
		trigger = { check_variable = { mdr_treaty_proposal = 4 } }
	}
	desc = {
		text = mdr_treaties.2.d5
		trigger = { check_variable = { mdr_treaty_proposal = 5 } }
	}
	desc = {
		text = mdr_treaties.2.d10
		trigger = { check_variable = { mdr_treaty_proposal = 10 } }
	}
	picture = GFX_message_signing
	is_triggered_only = yes

	option = {
		name = mdr_treaties.2.a
		log = "[GetDateText]: [This.GetName]: mdr_treaties.2.a executed"
		set_temp_variable = { mdr_treaty_id = mdr_treaty_proposal }
		if = { limit = { NOT = { check_variable = { mdr_treaty_proposal = 10 } } }
			mdr_treaty_join = yes
		}
		FROM = { mdr_treaty_join = yes }
		ai_chance = { base = 1 }
	}

	option = {
		name = mdr_treaties.2.b
		log = "[GetDateText]: [This.GetName]: mdr_treaties.2.b executed"
		FROM = { add_opinion_modifier = { target = ROOT modifier = mdr_left_arms_treaty } }
		ai_chance = {
			base = 1
			modifier = {
				factor = 3
				has_opinion = { target = FROM value < 0 }
			}
		}
	}
}
